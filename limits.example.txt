# Прайс-лист лимитов покупки по атрибутам подарка.
# Формат: тип:название=макс_цена_в_звёздах
# Типы: model (модель), backdrop (фон), pattern (узор/скин)
# Названия — точно как в Telegram (регистр не важен). Узнать: --list и вывод парсера.
#
# Как работает: у подарка есть модель + фон + узор. Если для подарка
# подходит несколько правил, цена должна быть НИЖЕ ВСЕХ подходящих
# (побеждает самый строгий лимит). Пустые строки и строки с # игнорируются.
#
# Чтобы включить: переименуйте этот файл в limits.txt
# (или укажите путь: limits=мой_файл.txt в config.txt)

# ── Лимиты по модели ──
model:Plush Pepe=500
model:Ocean Oasis=300
model:Heart Locket=180

# ── Лимиты по фону (backdrop) ──
backdrop:Lemongrass=250
backdrop:Black=400

# ── Лимиты по узору (pattern) ──
pattern:Camo=200
