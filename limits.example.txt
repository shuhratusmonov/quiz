# Прайс-лист лимитов покупки. Одна строка = одно правило: условия = макс_цена (⭐)
#
# Условия (можно комбинировать через +):
#   gift:НАЗВАНИЕ      — название подарка/коллекции (напр. Whip Cupcake)
#   model:НАЗВАНИЕ     — атрибут "модель"           (напр. Resistant)
#   backdrop:НАЗВАНИЕ  — фон                          (напр. Black)
#   pattern:НАЗВАНИЕ   — узор/скин                    (напр. Camo)
#
# Если в правиле несколько условий (через +), должны совпасть ВСЕ.
# Если подарку подходит несколько правил — берётся самый строгий (мин. цена).
# Названия — как показывает парсер (регистр не важен).
#
# Чтобы включить: переименуйте этот файл в limits.txt
# (или укажите путь: limits=мой_файл.txt в config.txt)

# ── Пример из вопроса: Whip Cupcake с моделью Resistant до 700 ⭐ ──
gift:Whip Cupcake+model:Resistant=700

# ── Любой подарок с конкретной моделью ──
model:Plush=1200

# ── По названию подарка целиком ──
gift:Heart Locket=180

# ── Комбинация подарок + фон ──
gift:Whip Cupcake+backdrop:Black=500

# ── По фону / узору отдельно ──
backdrop:Lemongrass=250
pattern:Camo=200
